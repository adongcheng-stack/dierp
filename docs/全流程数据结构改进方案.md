### 1.1 创建本厂编号映射表

```sql
-- 创建本厂编号映射表
CREATE TABLE factory_product_mappings (
    id SERIAL PRIMARY KEY,
    customer_id VARCHAR(50) NOT NULL REFERENCES customers(customer_id),
    factory_product_number VARCHAR(50) NOT NULL,
    material_code VARCHAR(50) NOT NULL REFERENCES materials(material_code),
    production_version_id VARCHAR(50) REFERENCES production_versions(production_version_id),
    is_default BOOLEAN DEFAULT TRUE,
    effective_date TIMESTAMP NOT NULL,
    expiry_date TIMESTAMP,
    created_by VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_by VARCHAR(50),
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    
    -- 唯一约束：同一客户的本厂编号在同一时间段内唯一
    CONSTRAINT uq_factory_product UNIQUE (customer_id, factory_product_number, effective_date)
);

-- 创建索引
CREATE INDEX idx_factory_product_mapping_customer ON factory_product_mappings(customer_id);
CREATE INDEX idx_factory_product_mapping_product ON factory_product_mappings(factory_product_number);
CREATE INDEX idx_factory_product_mapping_material ON factory_product_mappings(material_code);
CREATE INDEX idx_factory_product_mapping_version ON factory_product_mappings(production_version_id);
```

```python
class FactoryProductMapping(BaseModel):
    """本厂编号映射表"""
    __tablename__ = "factory_product_mappings"
    
    id = Column(Integer, primary_key=True)
    customer_id = Column(String(50), ForeignKey("customers.customer_id"), nullable=False)
    factory_product_number = Column(String(50), nullable=False)
    material_code = Column(String(50), ForeignKey("materials.material_code"), nullable=False)
    production_version_id = Column(String(50), ForeignKey("production_versions.production_version_id"))
    is_default = Column(Boolean, default=True)
    effective_date = Column(DateTime, nullable=False)
    expiry_date = Column(DateTime)
    
    # 关联
    customer = relationship("Customer")
    material = relationship("Material")
    production_version = relationship("ProductionVersion")
    
    __table_args__ = (
        UniqueConstraint('customer_id', 'factory_product_number', 'effective_date', name='uq_factory_product'),
    )
```

## 三、业务逻辑改进

### 3.1 报价单转销售订单逻辑

```python
# backend/api/routers/sd.py

async def create_sales_order_from_quotation(
    quotation_number: str,
    db: Session = Depends(get_db)
):
    """从报价单创建销售订单（自动映射本厂编号）"""
    quotation = db.query(Quotation).filter(
        Quotation.quotation_number == quotation_number
    ).first()
    
    if not quotation:
        raise HTTPException(status_code=404, detail="报价单不存在")
    
    # 查找本厂编号映射
    mapping = db.query(FactoryProductMapping).filter(
        FactoryProductMapping.customer_id == quotation.customer_id,
        FactoryProductMapping.factory_product_number == quotation.product_number,
        FactoryProductMapping.effective_date <= datetime.now(),
        or_(
            FactoryProductMapping.expiry_date.is_(None),
            FactoryProductMapping.expiry_date >= datetime.now()
        ),
        FactoryProductMapping.is_default == True
    ).first()
    
    if not mapping:
        raise HTTPException(
            status_code=400,
            detail=f"未找到本厂编号映射: {quotation.product_number}"
        )
    
    # 验证物料状态
    material = db.query(Material).filter(
        Material.material_code == mapping.material_code
    ).first()
    
    if not material or material.status != "ACTIVE":
        raise HTTPException(
            status_code=400,
            detail=f"物料不存在或未激活: {mapping.material_code}"
        )
    
    # 验证生产版本状态
    if mapping.production_version_id:
        pv = db.query(ProductionVersion).filter(
            ProductionVersion.production_version_id == mapping.production_version_id
        ).first()
        
        if not pv or pv.status not in ["RELEASED", "ACTIVE"]:
            raise HTTPException(
                status_code=400,
                detail=f"生产版本不可用: {mapping.production_version_id}"
            )
    
    # 计算生产数量和单位转换
    production_quantity = quotation.quantity
    production_unit = material.production_unit or material.base_unit
    sales_unit = material.base_unit
    conversion_factor = material.conversion_factor or 1
    
    if sales_unit != production_unit:
        # 需要单位转换
        production_quantity = quotation.quantity / conversion_factor
    
    # 创建销售订单
    sales_order = SalesOrder(
        order_number=generate_order_number(),
        quotation_number=quotation_number,
        customer_id=quotation.customer_id,
        product_number=quotation.product_number,
        material_code=mapping.material_code,
        production_version_id=mapping.production_version_id,
        quantity=quotation.quantity,
        sales_unit=sales_unit,
        production_quantity=production_quantity,
        production_unit=production_unit,
        conversion_factor=conversion_factor,
        unit_price=quotation.unit_price,
        total_amount=quotation.total_amount,
        currency=quotation.currency,
        delivery_date=calculate_delivery_date(quotation),
        status=SalesOrderStatus.DRAFT
    )
    
    db.add(sales_order)
    db.commit()
    return sales_order
```

### 3.2 销售订单转生产订单逻辑

```python
# backend/api/routers/pp.py

async def create_production_order_from_sales_order(
    sales_order_number: str,
    db: Session = Depends(get_db)
):
    """从销售订单创建生产订单（自动使用生产版本）"""
    sales_order = db.query(SalesOrder).filter(
        SalesOrder.order_number == sales_order_number
    ).first()
    
    if not sales_order:
        raise HTTPException(status_code=404, detail="销售订单不存在")
    
    if sales_order.status != SalesOrderStatus.APPROVED:
        raise HTTPException(
            status_code=400,
            detail="销售订单必须已审批才能创建生产订单"
        )
    
    if not sales_order.material_code:
        raise HTTPException(
            status_code=400,
            detail="销售订单缺少物料编码"
        )
    
    if not sales_order.production_version_id:
        raise HTTPException(
            status_code=400,
            detail="销售订单缺少生产版本"
        )
    
    # 验证生产版本
    pv = db.query(ProductionVersion).filter(
        ProductionVersion.production_version_id == sales_order.production_version_id
    ).first()
    
    if not pv:
        raise HTTPException(status_code=404, detail="生产版本不存在")
    
    if pv.status not in ["RELEASED", "ACTIVE"]:
        raise HTTPException(
            status_code=400,
            detail=f"生产版本状态不可用: {pv.status}"
        )
    
    # 验证物料编码一致性
    if pv.material_code != sales_order.material_code:
        raise HTTPException(
            status_code=400,
            detail="生产版本的物料编码与销售订单不一致"
        )
    
    # 创建生产订单
    production_order = ProductionOrder(
        order_number=generate_production_order_number(),
        sales_order_number=sales_order_number,
        material_code=sales_order.material_code,
        production_version_id=sales_order.production_version_id,
        quantity=sales_order.production_quantity or sales_order.quantity,
        unit=sales_order.production_unit or sales_order.sales_unit,
        sales_quantity=sales_order.quantity,
        sales_unit=sales_order.sales_unit,
        conversion_factor=sales_order.conversion_factor or 1,
        planned_start_date=calculate_planned_start_date(sales_order),
        planned_finish_date=calculate_planned_finish_date(sales_order),
        status=ProductionOrderStatus.DRAFT
    )
    
    db.add(production_order)
    db.commit()
    return production_order
```

## 四、API端点改进

### 4.1 本厂编号映射管理API

```python
# backend/api/routers/sd.py

@router.post("/factory-product-mappings", response_model=FactoryProductMappingResponse)
async def create_factory_product_mapping(
    mapping: FactoryProductMappingCreate,
    db: Session = Depends(get_db),
    current_user: User = Depends(get_current_user)
):
    """创建本厂编号映射"""
    # 验证客户存在
    customer = db.query(Customer).filter(
        Customer.customer_id == mapping.customer_id
    ).first()
    if not customer:
        raise HTTPException(status_code=404, detail="客户不存在")
    
    # 验证物料存在
    material = db.query(Material).filter(
        Material.material_code == mapping.material_code
    ).first()
    if not material:
        raise HTTPException(status_code=404, detail="物料不存在")
    
    # 验证生产版本（如果提供）
    if mapping.production_version_id:
        pv = db.query(ProductionVersion).filter(
            ProductionVersion.production_version_id == mapping.production_version_id
        ).first()
        if not pv:
            raise HTTPException(status_code=404, detail="生产版本不存在")
        if pv.material_code != mapping.material_code:
            raise HTTPException(
                status_code=400,
                detail="生产版本的物料编码与映射不一致"
            )
    
    # 如果设置为默认，取消其他默认映射
    if mapping.is_default:
        db.query(FactoryProductMapping).filter(
            FactoryProductMapping.customer_id == mapping.customer_id,
            FactoryProductMapping.factory_product_number == mapping.factory_product_number,
            FactoryProductMapping.is_default == True
        ).update({"is_default": False})
    
    db_mapping = FactoryProductMapping(
        **mapping.dict(),
        created_by=current_user.username
    )
    db.add(db_mapping)
    db.commit()
    return db_mapping


@router.get("/factory-product-mappings/{customer_id}/{product_number}")
async def get_product_mapping(
    customer_id: str,
    product_number: str,
    db: Session = Depends(get_db)
):
    """获取本厂编号映射"""
    mapping = db.query(FactoryProductMapping).filter(
        FactoryProductMapping.customer_id == customer_id,
        FactoryProductMapping.factory_product_number == product_number,
        FactoryProductMapping.effective_date <= datetime.now(),
        or_(
            FactoryProductMapping.expiry_date.is_(None),
            FactoryProductMapping.expiry_date >= datetime.now()
        ),
        FactoryProductMapping.is_default == True
    ).first()
    
    if not mapping:
        raise HTTPException(status_code=404, detail="未找到本厂编号映射")
    
    return mapping
```
