## 三、迁移详情

### 3.1 SalesOrderForm.tsx 迁移

#### 迁移前

```typescript
// 需要手动管理客户列表
const [customers, setCustomers] = useState<any[]>([])

useEffect(() => {
  fetchCustomers()
}, [])

const fetchCustomers = async () => {
  try {
    const response = await api.get('/sd/customers')
    setCustomers(response.data)
  } catch (error) {
    console.error('获取客户列表失败:', error)
  }
}

// 手动渲染选择框
<Select placeholder="请选择客户" showSearch disabled={fromQuotation}>
  {customers.map(customer => (
    <Option key={customer.customer_id} value={customer.customer_id}>
      {customer.legal_name} ({customer.customer_id})
    </Option>
  ))}
</Select>

// 手动管理按钮
<Space>
  <Button type="primary" htmlType="submit" loading={loading}>
    {isEdit ? '更新' : '创建'}
  </Button>
  <Button onClick={() => navigate('/sd/sales-orders')}>
    取消
  </Button>
</Space>
```

#### 迁移后

```typescript
// 直接使用共享组件
import CustomerSelect from '../../components/common/CustomerSelect'
import FormButtons from '../../components/common/FormButtons'
import { handleApiError } from '../../utils/errorHandler'

<CustomerSelect disabled={fromQuotation} />

<FormButtons
  loading={loading}
  onSave={() => form.submit()}
  saveText={isEdit ? '更新' : '创建'}
  cancelPath="/sd/sales-orders"
/>

// 统一错误处理
catch (error: any) {
  handleApiError(error)
}
```

**优势：**
- ✅ 减少约30行代码
- ✅ 自动处理加载状态
- ✅ 统一的搜索和过滤逻辑
- ✅ 统一的错误处理

### 6.1 迁移步骤
